seconds def clean_html_content(self, html_content: str) -> str: """清理HTML内容，移除以http://或https://开头的内容""" try: if not html_content: return "" # 移除以http://或https://开头的内容 cleaned_content = re.sub(r'https?://\S+', '', html_content) #
